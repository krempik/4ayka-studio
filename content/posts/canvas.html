Основа любой Canvas-игры — цикл requestAnimationFrame. Минимум теории, максимум кода.

Базовый цикл

function gameLoop(ts) {
    update(dt);   // логика
    draw();       // отрисовка
    requestAnimationFrame(gameLoop);
}
requestAnimationFrame(gameLoop);

Фиксированный шаг — правильно

Если физика на клиенте, шаг времени должен быть стабильным. Накапливаем время и выполняем симуляцию фиксированными тиками:

const STEP = 1000 / 30;
let acc = 0, last = 0;
function frame(ts) {
    acc += Math.min(ts - last, 250);
    last = ts;
    while (acc >= STEP) { update(STEP); acc -= STEP; }
    draw();
    requestAnimationFrame(frame);
}
  • Кап аккумулятора (250 мс) спасает от «спирали смерти» при фоне вкладки
  • Такой паттерн используется в Dungeon of the Void

Коллизии

function collides(a, b) {
    return a.x < b.x + b.w && a.x + a.w > b.x &&
           a.y < b.y + b.h && a.y + a.h > b.y;
}

DPR: чтобы не было мыла

На экранах с плотностью 2–3x растянутый канвас выглядит размытым. Учитывай devicePixelRatio и масштабируй контекст.

const dpr = Math.min(devicePixelRatio || 1, 2);
canvas.width = width * dpr;
canvas.height = height * dpr;
ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
